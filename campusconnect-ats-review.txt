const emptyProfile = {
  rollNumber: '',
  phone: '',
  dateOfBirth: '',
  degree: '',
  branch: '',
  graduationYear: '',
  cgpa: '',
  activeBacklogs: '0',
};

function toFormValues(profile) {
  return {
    rollNumber: profile.rollNumber ?? '',
    phone: profile.phone ?? '',
    dateOfBirth: profile.dateOfBirth ?? '',
    degree: profile.degree ?? '',
    branch: profile.branch ?? '',
    graduationYear: profile.graduationYear?.toString() ?? '',
    cgpa: profile.cgpa?.toString() ?? '',
    activeBacklogs: (profile.activeBacklogs ?? 0).toString(),
  };
}

function validateStudentProfile(form) {
  const year = form.graduationYear.trim();
  const cgpa = form.cgpa.trim();
  const backlogs = form.activeBacklogs.trim();

  if (form.rollNumber.length > 40) return 'Roll number must be 40 characters or fewer.';
  if (form.phone.length > 20) return 'Phone number must be 20 characters or fewer.';
  if (form.degree.length > 100) return 'Degree must be 100 characters or fewer.';
  if (form.branch.length > 120) return 'Branch must be 120 characters or fewer.';

  if (form.dateOfBirth && form.dateOfBirth >= new Date().toISOString().slice(0, 10)) {
    return 'Date of birth must be in the past.';
  }

  if (year && (!/^\d{4}$/.test(year) || Number(year) < 2000 || Number(year) > 2100)) {
    return 'Graduation year must be between 2000 and 2100.';
  }

  if (cgpa && (!/^\d{1,2}(\.\d{1,2})?$/.test(cgpa) || Number(cgpa) < 0 || Number(cgpa) > 10)) {
    return 'CGPA must be between 0 and 10, with up to two decimal places.';
  }

  if (!/^\d+$/.test(backlogs) || Number(backlogs) > 32767) {
    return 'Active backlogs must be a non-negative whole number.';
  }

  return '';
}

  const [pwMsg, setPwMsg] = useState('');

      dateOfBirth: profile.dateOfBirth || null,
      degree: profile.degree.trim() || null,
      branch: profile.branch.trim() || null,
      graduationYear: profile.graduationYear ? Number(profile.graduationYear) : null,
      cgpa: profile.cgpa ? Number(profile.cgpa) : null,

      setProfileSaving(false);

package com.campusconnect.student;

import com.campusconnect.auth.AuthUser;
import com.campusconnect.common.ApiException;
import java.io.IOException;
import java.util.Locale;
import org.apache.pdfbox.Loader;
import org.apache.pdfbox.pdmodel.PDDocument;
import org.apache.pdfbox.text.PDFTextStripper;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;

import static com.campusconnect.student.StudentProfileDtos.*;

@Service
public class StudentProfileService {

    private static final long MAX_RESUME_SIZE = 5 * 1024 * 1024;

    private final StudentProfileRepository profiles;

    public StudentProfileService(StudentProfileRepository profiles) {
        this.profiles = profiles;
    }

    @Transactional(readOnly = true)
    public StudentProfileResponse getMyProfile(AuthUser me) {
        return profiles.findByUserId(me.id())
                .map(StudentProfileResponse::from)
                .orElseThrow(() -> new ApiException(
                        HttpStatus.NOT_FOUND,
                        "Student profile not found. Save your profile first."));
    }

    @Transactional
    public StudentProfileResponse updateMyProfile(
            AuthUser me, UpdateStudentProfileRequest request) {

        StudentProfile profile = profiles.findByUserId(me.id())
                .orElseGet(() -> new StudentProfile(me.id()));

        String rollNumber = normalize(request.rollNumber());

        if (rollNumber != null
                && profiles.existsByRollNumberIgnoreCase(rollNumber)
                && !rollNumber.equalsIgnoreCase(profile.getRollNumber())) {
            throw new ApiException(
                    HttpStatus.CONFLICT,
                    "That roll number is already in use.");
        }

        profile.updateDetails(
                rollNumber,
                normalize(request.phone()),
                request.dateOfBirth(),
                normalize(request.degree()),
                normalize(request.branch()),
                request.graduationYear(),
                request.cgpa(),
                request.activeBacklogs(),
                normalize(request.resumeUrl()));

        try {
            return StudentProfileResponse.from(
                    profiles.saveAndFlush(profile));
        } catch (DataIntegrityViolationException ex) {
            throw new ApiException(
                    HttpStatus.CONFLICT,
                    "The profile conflicts with an existing record.");
        }
    }

    @Transactional
    public void uploadResume(AuthUser me, MultipartFile file) {
        if (file == null || file.isEmpty()) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST, "Please select a PDF resume.");
        }

        if (file.getSize() > MAX_RESUME_SIZE) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST, "Resume must be 5 MB or smaller.");
        }

        String contentType = file.getContentType();
        if (!"application/pdf".equalsIgnoreCase(contentType)) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST, "Only PDF resumes are allowed.");
        }

        try {
            byte[] data = file.getBytes();

            // Validate that the uploaded file is actually a readable PDF.
            try (PDDocument document = Loader.loadPDF(data)) {
                if (document.isEncrypted()) {
                    throw new ApiException(
                            HttpStatus.BAD_REQUEST,
                            "Password-protected PDFs are not supported.");
                }
            }

            StudentProfile profile = profiles.findByUserId(me.id())
                    .orElseGet(() -> new StudentProfile(me.id()));

            String fileName = file.getOriginalFilename();
            if (fileName == null || fileName.isBlank()) {
                fileName = "resume.pdf";
            }

            profile.storeResume(fileName, "application/pdf", data);
            profiles.save(profile);

        } catch (IOException ex) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST, "The uploaded PDF could not be read.");
        }
    }

    @Transactional(readOnly = true)
    public ResumeFile getMyResume(AuthUser me) {
        StudentProfile profile = profiles.findByUserId(me.id())
                .orElseThrow(() -> new ApiException(
                        HttpStatus.NOT_FOUND, "Resume not found."));

        if (!profile.hasResume()) {
            throw new ApiException(
                    HttpStatus.NOT_FOUND, "Resume not found. Upload a PDF first.");
        }

        return new ResumeFile(
                profile.getResumeFileName(),
                profile.getResumeData());
    }

    @Transactional
    public void deleteMyResume(AuthUser me) {
        StudentProfile profile = profiles.findByUserId(me.id())
                .orElseThrow(() -> new ApiException(
                        HttpStatus.NOT_FOUND, "Student profile not found."));

        profile.removeResume();
        profiles.save(profile);
    }

    @Transactional(readOnly = true)
    public ResumeAnalysis analyzeResume(AuthUser me, String jobDescription) {
        if (jobDescription == null || jobDescription.isBlank()) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST,
                    "Please provide a job description.");
        }

        if (jobDescription.length() > 20000) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST,
                    "Job description must be 20,000 characters or fewer.");
        }

        StudentProfile profile = profiles.findByUserId(me.id())
                .orElseThrow(() -> new ApiException(
                        HttpStatus.NOT_FOUND, "Student profile not found."));

        if (!profile.hasResume()) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST,
                    "Upload your PDF resume before analyzing it.");
        }

        try (PDDocument document = Loader.loadPDF(profile.getResumeData())) {
            String resumeText = new PDFTextStripper().getText(document);
            if (resumeText.isBlank()) {
                throw new ApiException(
                        HttpStatus.BAD_REQUEST,
                        "No readable text was found in the PDF. Please upload a text-based PDF.");
            }

            var resumeWords = extractKeywords(resumeText);
            var jobWords = extractKeywords(jobDescription);

            var matched = jobWords.stream()
                    .filter(resumeWords::contains)
                    .sorted()
                    .toList();

            var missing = jobWords.stream()
                    .filter(word -> !resumeWords.contains(word))
                    .sorted()
                    .toList();

            int score = jobWords.isEmpty()
                    ? 0
                    : (int) Math.round(100.0 * matched.size() / jobWords.size());

            return new ResumeAnalysis(
                    score, matched, missing,
                    "This is a keyword-based estimate, not a guarantee of ATS performance.");
        } catch (IOException ex) {
            throw new ApiException(
                    HttpStatus.BAD_REQUEST, "Could not read the stored PDF resume.");
        }
    }

    private java.util.Set<String> extractKeywords(String text) {
        return java.util.Arrays.stream(
                        text.toLowerCase(Locale.ROOT)
                                .replaceAll("[^a-z0-9+#.]", " ")
                                .split("\\s+"))
                .filter(word -> word.length() >= 3)
                .filter(word -> !java.util.Set.of(
                        "the", "and", "for", "with", "you", "your",
                        "are", "our", "this", "that", "will", "from",
                        "have", "has", "can", "all", "any", "job",
                        "role", "work", "years", "experience").contains(word))
                .collect(java.util.stream.Collectors.toSet());
    }

    private String normalize(String value) {
        if (value == null) {
            return null;
        }
        String trimmed = value.trim();
        return trimmed.isEmpty() ? null : trimmed;
    }

    public record ResumeFile(String fileName, byte[] data) {
    }

    public record ResumeAnalysis(
            int score,
            java.util.List<String> matchedKeywords,
            java.util.List<String> missingKeywords,
            String disclaimer) {
    }
}

package com.campusconnect.student;

import com.campusconnect.auth.AuthUser;
import com.campusconnect.student.StudentProfileDtos.StudentProfileResponse;
import com.campusconnect.student.StudentProfileDtos.UpdateStudentProfileRequest;
import jakarta.validation.Valid;
import org.springframework.http.ContentDisposition;
import org.springframework.http.HttpHeaders;
import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

@RestController
@RequestMapping("/api/v1/students/me")
@PreAuthorize("hasRole('STUDENT')")
public class StudentProfileController {

    private final StudentProfileService profiles;

    public StudentProfileController(StudentProfileService profiles) {
        this.profiles = profiles;
    }

    @GetMapping
    public StudentProfileResponse getMyProfile(
            @AuthenticationPrincipal AuthUser me) {
        return profiles.getMyProfile(me);
    }

    @PutMapping
    public StudentProfileResponse updateMyProfile(
            @AuthenticationPrincipal AuthUser me,
            @Valid @RequestBody UpdateStudentProfileRequest request) {
        return profiles.updateMyProfile(me, request);
    }

    @PostMapping(value = "/resume", consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
    public ResponseEntity<Void> uploadResume(
            @AuthenticationPrincipal AuthUser me,
            @RequestParam("file") MultipartFile file) {
        profiles.uploadResume(me, file);
        return ResponseEntity.noContent().build();
    }

    @GetMapping("/resume")
    public ResponseEntity<byte[]> downloadResume(
            @AuthenticationPrincipal AuthUser me) {
        StudentProfileService.ResumeFile resume = profiles.getMyResume(me);

        return ResponseEntity.ok()
                .contentType(MediaType.APPLICATION_PDF)
                .header(HttpHeaders.CONTENT_DISPOSITION,
                        ContentDisposition.attachment()
                                .filename(resume.fileName())
                                .build()
                                .toString())
                .body(resume.data());
    }

    @DeleteMapping("/resume")
    public ResponseEntity<Void> deleteResume(
            @AuthenticationPrincipal AuthUser me) {
        profiles.deleteMyResume(me);
        return ResponseEntity.noContent().build();
    }

    @PostMapping("/resume/analyze")
    public StudentProfileService.ResumeAnalysis analyzeResume(

        return profiles.analyzeResume(me, request.jobDescription());

package com.campusconnect.student;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.Instant;

@Entity
@Table(name = "student_profiles")
public class StudentProfile {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "user_id", nullable = false, unique = true)
    private Long userId;

    @Column(name = "roll_number", length = 40, unique = true)
    private String rollNumber;

    @Column(length = 20)
    private String phone;

    @Column(name = "date_of_birth")
    private LocalDate dateOfBirth;

    @Column(length = 100)
    private String degree;

    @Column(length = 120)
    private String branch;

    @Column(name = "graduation_year")
    private Short graduationYear;

    @Column(precision = 4, scale = 2)
    private BigDecimal cgpa;

    @Column(name = "active_backlogs", nullable = false)
    private Short activeBacklogs = 0;

    // Legacy resume URL retained for compatibility
    @Column(name = "resume_url", length = 2048)
    private String resumeUrl;

    // Resume PDF storage
    @Column(name = "resume_file_name")
    private String resumeFileName;

    @Column(name = "resume_content_type")
    private String resumeContentType;

    @Column(name = "resume_data")
    private byte[] resumeData;

    @Column(name = "resume_uploaded_at")
    private Instant resumeUploadedAt;

    protected StudentProfile() {
    }

    public StudentProfile(Long userId) {
        this.userId = userId;
        this.activeBacklogs = 0;
    }

    public Long getId() {
        return id;
    }

    public Long getUserId() {
        return userId;
    }

    public String getRollNumber() {
        return rollNumber;
    }

    public String getPhone() {
        return phone;
    }

    public LocalDate getDateOfBirth() {
        return dateOfBirth;
    }

    public String getDegree() {
        return degree;
    }

    public String getBranch() {
        return branch;
    }

    public Short getGraduationYear() {
        return graduationYear;
    }

    public BigDecimal getCgpa() {
        return cgpa;
    }

    public Short getActiveBacklogs() {
        return activeBacklogs;
    }

    public String getResumeUrl() {
        return resumeUrl;
    }

    public String getResumeFileName() {
        return resumeFileName;
    }

    public String getResumeContentType() {
        return resumeContentType;
    }

    public byte[] getResumeData() {
        return resumeData;
    }

    public Instant getResumeUploadedAt() {
        return resumeUploadedAt;
    }

    public boolean hasResume() {
        return resumeData != null && resumeData.length > 0;
    }

    public void storeResume(
            String fileName,
            String contentType,
            byte[] data) {

        this.resumeFileName = fileName;
        this.resumeContentType = contentType;
        this.resumeData = data;
        this.resumeUploadedAt = Instant.now();

        // New resume uploads supersede the old URL.
        this.resumeUrl = null;
    }

    public void removeResume() {
        this.resumeFileName = null;
        this.resumeContentType = null;
        this.resumeData = null;
        this.resumeUploadedAt = null;
        this.resumeUrl = null;
    }

    public void updateDetails(
            String rollNumber,
            String phone,
            LocalDate dateOfBirth,
            String degree,
            String branch,
            Short graduationYear,
            BigDecimal cgpa,
            Short activeBacklogs,
            String resumeUrl) {

        this.rollNumber = rollNumber;
        this.phone = phone;
        this.dateOfBirth = dateOfBirth;
        this.degree = degree;
        this.branch = branch;
        this.graduationYear = graduationYear;
        this.cgpa = cgpa;
        this.activeBacklogs = activeBacklogs == null ? 0 : activeBacklogs;

        // Preserve uploaded resume; only update the legacy URL if provided.
        if (resumeData == null) {
            this.resumeUrl = resumeUrl;
        }
    }
}

public final class StudentProfileDtos {

    }

    public record UpdateStudentProfileRequest(
            @Size(max = 40) String rollNumber,
            @Size(max = 20) String phone,
            @Past LocalDate dateOfBirth,
            @Size(max = 100) String degree,
            @Size(max = 120) String branch,
            @Min(2000) @Max(2100) Short graduationYear,
            @DecimalMin("0.00") @DecimalMax("10.00") BigDecimal cgpa,
            @PositiveOrZero Short activeBacklogs,
            @Size(max = 2048) String resumeUrl
    ) {
    }

            String rollNumber,
            String phone,

                    p.getDegree(),
                    p.getBranch(),
                    p.getGraduationYear(),
                    p.getCgpa(),
                    p.getActiveBacklogs(),
